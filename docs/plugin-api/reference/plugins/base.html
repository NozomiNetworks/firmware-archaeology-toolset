plugins.base

Shared plugin framework for the Firmware Archaeology Toolset.

This module defines the core extension surface used by FAT plugins:

  • PluginContext carries normalized runtime settings into plugin code.
  • SurfacePluginBase implements the standard stateful workflow used by most source plugins.

Plugin authors should usually start with SurfacePluginBase and override only run() and download(). More complex plugins may also override parser, validation, or command handlers, but the default path is intentionally small.

 118    def queries(self) -> list[str]:
 119        """Return a stable, deduplicated list of query strings.
 120
 121        Standard surface plugins search once per keyword. This helper removes
 122        blanks and preserves the first occurrence ordering so that reports and
 123        run history stay deterministic.
 124        """
 125
 126        queries: list[str] = []
 127        for keyword in self.keywords:

 234def keyword_variants(keyword: str) -> list[str]:
 235    base = normalize(keyword)
 236    compact = base.replace(" ", "")
 237    variants = {base, compact}
 238    for suffix in ("sdk", "api", "ai"):
 239        if compact.endswith(suffix) and len(compact) > len(suffix) + 1:
 240            variants.add(f"{compact[:-len(suffix)]} {suffix}")
 241    return [variant for variant in variants if variant]

 244def matched_keywords(text: str, keywords: list[str]) -> list[str]:
 245    haystack = normalize(text)
 246    token_set = set(haystack.split())
 247    compact_haystack = haystack.replace(" ", "")
 248    matched = []
 249    for keyword in keywords:
 250        for variant in keyword_variants(keyword):
 251            if " " in variant:
 252                variant_tokens = variant.split()
 253                if variant in haystack or all(token in token_set for token in variant_tokens):
 254                    matched.append(keyword)
 255                    break
 256            else:
 257                if variant in token_set or variant in compact_haystack:
 258                    matched.append(keyword)
 259                    break
 260    return matched

 263def classify_match(
 264    keywords: list[str],
 265    min_match_keywords: int,
 266    name: str,
 267    description: str,
 268    extra_text: str = "",
 269) -> tuple[str | None, list[str], list[str]]:
 270    """Classify one candidate artifact against the project's keyword set.
 271
 272    Returns a tuple containing:
 273
 274    - a confidence label such as `high` or `review`
 275    - the matched project keywords
 276    - normalized signals explaining why the candidate matched
 277    """
 278
 279    haystack = f"{name} {description} {extra_text}"
 280    hits = matched_keywords(haystack, keywords)
 281    signals: list[str] = []
 282
 283    if hits:
 284        signals.extend(f"keyword:{keyword}" for keyword in hits)
 285    if extra_text.strip():
 286        signals.append("extra-text")
 287
 288    if len(hits) >= min_match_keywords:
 289        return "high", hits, signals
 290    if hits:
 291        return "review", hits, signals
 292    return None, hits, signals

 407def merge_named_results(results: list[dict[str, Any]]) -> list[dict[str, Any]]:
 408    """Merge duplicate artifact candidates by `name`.
 409
 410    Queries, matched keywords, and signals are unioned. The descriptive fields
 411    from the strongest individual match are kept, where strength is determined
 412    first by keyword-hit count and then by confidence label.
 413    """
 414
 415    confidence_rank = {"high": 3, "review": 2, "catalog": 1, "error": 0}
 416    merged: dict[str, dict[str, Any]] = {}
 417    for result in results:
 418        current = merged.get(result["name"])
 419        if current is None:
 420            result["matched_queries"] = [result["query"]]
 421            merged[result["name"]] = result
 422            continue
 423        current_hit_count = len(current.get("matched_keywords", []))
 424        result_hit_count = len(result.get("matched_keywords", []))
 425        result_is_stronger = (
 426            result_hit_count > current_hit_count
 427            or (
 428                result_hit_count == current_hit_count
 429                and confidence_rank.get(str(result.get("confidence")), -1) > confidence_rank.get(str(current.get("confidence")), -1)
 430            )
 431        )
 432        current["matched_keywords"] = sorted(set(current["matched_keywords"]) | set(result["matched_keywords"]))
 433        current["signals"] = sorted(set(current["signals"]) | set(result["signals"]))
 434        current["matched_queries"] = sorted(set(current["matched_queries"]) | {result["query"]})
 435        if result_is_stronger:
 436            current["description"] = result["description"]
 437            current["confidence"] = result["confidence"]
 438            current["url"] = result["url"]
 439    return sorted(merged.values(), key=lambda item: (-len(item["matched_keywords"]), item["name"]))

 442def resolve_selection(selection: str, artifacts: list[dict[str, Any]], selected_ids: set[str]) -> list[dict[str, Any]]:
 443    by_id, by_num = artifact_index(artifacts)
 444    if selection == "selected":
 445        return [by_id[artifact_id] for artifact_id in selected_ids if artifact_id in by_id]
 446    if selection == "all":
 447        return artifacts
 448    chosen: list[dict[str, Any]] = []
 449    for piece in selection.split(","):
 450        token = piece.strip()
 451        if not token:
 452            continue
 453        if token.isdigit() and int(token) in by_num:
 454            chosen.append(by_num[int(token)])
 455        elif token in by_id:
 456            chosen.append(by_id[token])
 457        else:
 458            raise SystemExit(f"Unknown artifact selection: {token}")
 459    return chosen

 663    def validate_args(self, args) -> None:
 664        """Validate the positional shape required by the selected command.
 665
 666        Override this together with `parse_args()` for non-standard workflows.
 667        """
 668
 669        command = args.command
 670        selection = args.selection
 671        value = args.value
 672
 673        if command in {"search", "status", "options", "clear-selection"}:
 674            require_no_extra_args(self.name, command, selection, value)
 675            return
 676
 677        if command == "select":
 678            require_selection_only(
 679                self.name,
 680                command,
 681                selection,
 682                value,
 683                message=f"`{self.name} select` requires an artifact selection like `1,2` and no extra trailing value.",
 684            )
 685            return
 686
 687        if command == "download":
 688            require_optional_selection(
 689                self.name,
 690                command,
 691                value,
 692                message=f"`{self.name} download` accepts at most one selector such as `selected`, `all`, or `1,2`.",
 693            )
 694            return
 695
 696        if command == "set":
 697            require_setting_assignment(self.name, selection, value)
 698            return

1154    def set(self, project_name: str, project: dict[str, Any], args) -> int:
1155        """Persist one plugin setting using the shared option model.
1156
1157        Supported settings in the base implementation are:
1158
1159        - `downloads-dir`
1160        - `per-query`
1161        - `min-match-keywords`
1162
1163        Subclasses may extend this command for plugin-specific settings and may
1164        honor shared unset tokens such as `None`, `null`, `""`, and `''`.
1165        """
1166
1167        if not args.selection:
1168            raise SystemExit(f"Missing setting name. Use `{self.name} set <name> <value>`.")
1169        if args.value is None:
1170            raise SystemExit(f"Missing setting value. Use `{self.name} set {args.selection} <value>`.")
1171        tool_state = self.ensure_state(project)
1172        setting = args.selection.replace("_", "-").lower()
1173        if setting == "keywords":
1174            raise SystemExit(
1175                f"`{self.name} set keywords` is not supported because keywords are project-level. "
1176                "Use `fat.py set keywords \"k1,k2\"` or `fat.py project set keywords \"k1,k2\"`."
1177            )
1178        unset_requested = is_unset_value(args.value)
1179        definitions = self.setting_map(project, tool_state)
1180        spec = definitions.get(setting)
1181        if spec is None:
1182            raise SystemExit(
1183                f"Unsupported setting for {self.name}: {args.selection}. "
1184                f"Use one of `{ '`, `'.join(sorted(definitions)) }`."
1185            )
1186        if unset_requested:

def merge_named_results(results: list[dict[str, typing.Any]]) -> list[dict[str, typing.Any]]: 
408def merge_named_results(results: list[dict[str, Any]]) -> list[dict[str, Any]]:
409    """Merge duplicate artifact candidates by `name`.
410
411    Queries, matched keywords, and signals are unioned. The descriptive fields
412    from the strongest individual match are kept, where strength is determined
413    first by keyword-hit count and then by confidence label.
414    """
415
416    confidence_rank = {"high": 3, "review": 2, "catalog": 1, "error": 0}
417    merged: dict[str, dict[str, Any]] = {}
418    for result in results:
419        current = merged.get(result["name"])
420        if current is None:
421            result["matched_queries"] = [result["query"]]
422            merged[result["name"]] = result
423            continue
424        current_hit_count = len(current.get("matched_keywords", []))
425        result_hit_count = len(result.get("matched_keywords", []))
426        result_is_stronger = (
427            result_hit_count > current_hit_count
428            or (
429                result_hit_count == current_hit_count
430                and confidence_rank.get(str(result.get("confidence")), -1) > confidence_rank.get(str(current.get("confidence")), -1)
431            )
432        )
433        current["matched_keywords"] = sorted(set(current["matched_keywords"]) | set(result["matched_keywords"]))
434        current["signals"] = sorted(set(current["signals"]) | set(result["signals"]))
435        current["matched_queries"] = sorted(set(current["matched_queries"]) | {result["query"]})
436        if result_is_stronger:
437            current["description"] = result["description"]
438            current["confidence"] = result["confidence"]
439            current["url"] = result["url"]
440    return sorted(merged.values(), key=lambda item: (-len(item["matched_keywords"]), item["name"]))
