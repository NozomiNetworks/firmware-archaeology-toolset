 30class HrefCollector(HTMLParser):
 31    """@private
 32
 33    Minimal HTML parser that extracts `href` attribute values.
 34    """
 35
 36    def __init__(self) -> None:
 37        super().__init__()
 38        self.hrefs: list[str] = []
 39
 40    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
 41        for key, value in attrs:
 42            if key.lower() == "href" and value:
 43                self.hrefs.append(value)

 46class WebsiteScraperBase(ABC):
 47    """Base class for one website-specific scraper implementation.
 48
 49    A website scraper is not a top-level FAT plugin. Instead, it is selected by
 50    the `scraper` plugin to handle one marketplace or download surface.
 51
 52    Subclasses should provide:
 53
 54    - `name`
 55    - `help_summary`
 56    - `default_website` when there is a sensible default entrypoint
 57    - `can_handle()` for parser auto-detection
 58    - `extract_product_links()`
 59    - `parse_product()`
 60    """
 61
 62    name = ""
 63    help_summary = ""
 64    default_website = ""
 65    # Rendering is deliberately opt-in.  Existing scrapers continue to use
 66    # the inexpensive static HTTP path unless they declare otherwise.
 67    RENDER_MODE = "static"
 68    RENDER_WAIT_UNTIL = "domcontentloaded"
 69    RENDER_WAIT_FOR: str | None = None
 70    RENDER_TIMEOUT: int | None = None
 71
 72    def can_handle(self, website_url: str, entry_html: str = "") -> bool:
 73        """Return `True` when this scraper can parse the target website."""
 74
 75        return False

 77    def fetch_text(self, url: str, timeout: int) -> str:
 78        """Fetch one HTML page for scraper-side parsing."""
 79
 80        return shared_fetch_text(url, timeout=timeout, accept="text/html,application/xhtml+xml")
 81
 82    def browser_options(self, timeout: int) -> BrowserOptions:
 83        """Return the validated browser settings declared by this scraper."""
 84
 85        return BrowserOptions(
 86            mode=str(self.RENDER_MODE),
 87            wait_until=str(self.RENDER_WAIT_UNTIL),
 88            wait_for=self.RENDER_WAIT_FOR,
 89            timeout=int(self.RENDER_TIMEOUT or timeout),
 90        )
 91
 92    def fetch_rendered_text(
 93        self,
 94        url: str,
 95        timeout: int,
 96        *,
 97        session: BrowserSession | None = None,
 98    ) -> str:
 99        """Fetch browser-rendered HTML, optionally reusing a crawl session."""
100
101        options = self.browser_options(timeout)
102        if session is not None:
103            return session.render(url, timeout=options.timeout)
104        with BrowserSession(options) as browser:
105            return browser.render(url, timeout=options.timeout)

107    def _fetch_page_text(
108        self,
109        url: str,
110        timeout: int,
111        *,
112        session: BrowserSession | None = None,
113        verbose: bool = False,
114        browser_disabled: bool = False,
115    ) -> str:
116        """Fetch one parser input according to the scraper's render mode."""
117
118        mode = str(self.RENDER_MODE)
119        if mode == "static" or browser_disabled:
120            return self.fetch_text(url, timeout)
121        try:
122            return self.fetch_rendered_text(url, timeout, session=session)
123        except BrowserError:
124            if mode != "preferred":
125                raise
126            if verbose:
127                print(f"[!] [scraper:{self.name}] browser rendering failed; falling back to static HTML")
128            return self.fetch_text(url, timeout)

131        """Download one remote file into a local destination path."""

133        shared_download_file(url, destination, timeout=timeout)

136        """Extract raw `href` values from an HTML document."""

142    def same_origin(self, url: str, base_url: str) -> bool:
143        """Return whether two URLs have the same HTTP(S) scheme and network location."""
144
145        try:
146            parsed = urllib.parse.urlsplit(url)
147            base = urllib.parse.urlsplit(base_url)
148            parsed_scheme = parsed.scheme.lower()
149            base_scheme = base.scheme.lower()
150            return (
151                parsed_scheme in ALLOWED_URL_SCHEMES
152                and base_scheme in ALLOWED_URL_SCHEMES
153                and parsed_scheme == base_scheme
154                and parsed.netloc.lower() == base.netloc.lower()
155            )
156        except ValueError:
157            return False

159    def crawl(self, website_url: str, timeout: int, *, verbose: bool = False) -> dict[str, Any]:
160        """Crawl the configured category page and return a normalized catalog."""
161
162        session: BrowserSession | None = None
163        mode = str(self.RENDER_MODE)
164        browser_disabled = False
165        if mode != "static":
166            try:
167                session = BrowserSession(self.browser_options(timeout))
168                session.__enter__()
169            except BrowserError:
170                if mode != "preferred":
171                    raise
172                if verbose:
173                    print(f"[!] [scraper:{self.name}] browser unavailable; falling back to static HTML")
174                session = None
175                browser_disabled = True
176        try:
177            entry_html = self._fetch_page_text(
178                website_url,
179                timeout,
180                session=session,
181                verbose=verbose,
182                browser_disabled=browser_disabled,
183            )
184            product_urls = self.extract_product_links(entry_html, website_url)
185            applications: list[dict[str, Any]] = []
186            for product_url in product_urls:
187                if verbose:
188                    print(f"[*] [scraper:{self.name}] fetching {product_url}")
189                product_html = self._fetch_page_text(
190                    product_url,
191                    timeout,
192                    session=session,
193                    verbose=verbose,
194                    browser_disabled=browser_disabled,
195                )
196                product = self.parse_product(product_html, product_url)
197                applications.append(product)
198            return self.build_catalog(website_url, applications)
199        finally:
200            if session is not None:
201                session.close()
202
203    def build_catalog(self, website_url: str, applications: list[dict[str, Any]]) -> dict[str, Any]:
204        """Build the persisted catalog document for one scraper run."""
205
206        total_download_links = sum(len(app.get("download_urls", [])) for app in applications)
207        unique_download_ids = len(
208            {
209                download_id
210                for app in applications
211                for download_id in app.get("download_ids", [])
212            }
213        )
214        return {
215            "source": website_url,
216            "scraper": self.name,
217            "applications": applications,
218            "summary": {
219                "application_count": len(applications),
220                "applications_with_downloads": sum(1 for app in applications if app.get("download_urls")),
221                "total_download_links": total_download_links,
222                "unique_download_ids": unique_download_ids,
223            },
224        }
225
226    def product_artifact(self, product: dict[str, Any], website_url: str) -> dict[str, Any]:
227        """Convert one parsed product record into a FAT artifact entry."""
228
229        slug = str(product["slug"])
230        title = str(product["title"])
231        download_urls = list(product.get("download_urls", []))
232        return {
233            "plugin": "scraper",
234            "kind": "website_artifact_bundle",
235            "query": website_url,
236            "name": slug,
237            "url": str(product.get("product_url", website_url)),
238            "description": f"{title} ({len(download_urls)} download(s))",
239            "matched_keywords": [],
240            "signals": ["scraped", f"scraper:{self.name}"],
241            "confidence": "catalog",
242            "title": title,
243            "slug": slug,
244            "download_urls": download_urls,
245            "download_ids": list(product.get("download_ids", [])),
246            "saved_files": list(product.get("saved_files", [])),
247            "website": website_url,
248            "scraper_name": self.name,
249        }
250
251    def filename_from_download_url(self, download_url: str) -> str:
252        """Derive a local filename from one download URL."""
253
254        parsed = urllib.parse.urlsplit(download_url)
255        name = Path(parsed.path).name
256        if not name.strip("."):
257            download_id_from_url = getattr(self, "download_id_from_url", None)
258            if callable(download_id_from_url):
259                try:
260                    return f"download-{download_id_from_url(download_url)}.zip"
261                except (TypeError, ValueError):
262                    pass
263            return "download.bin"
264        return name
265
266    def relative_download_path(self, product_slug: str, download_url: str) -> Path:
267        """Return the relative on-disk path used for one downloaded file."""
268
269        return Path(safe_name(product_slug)) / self.filename_from_download_url(download_url)

273        """Return product detail URLs discovered on the category page."""

279        """Parse one product page into the normalized product record schema."""
