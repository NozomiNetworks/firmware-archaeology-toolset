plugins.wayback.parsers.api

Documented extension contract for Wayback parser modules.

Wayback parser modules are lightweight adapters that teach the wayback plugin how to extract firmware download links from archived pages for one vendor or site family.

The implementation model is a plain module exporting:

  • extract_links(...) as the required parser function
  • update_firmware_link(...) as an optional pre-download hook
  • post_download(...) as an optional follow-on hook

Wayback parser authors usually need to understand four things:

  • the persisted Wayback project state shape
  • the per-snapshot record shape
  • the Capture record used by the archive client
  • the small subset of ArchiveClient methods that parsers can call safely

For replay-URL normalization, parsers should reuse:

  • ArchiveClient.unwrap_replay_url(...)

For parser setup and naming conventions, see:

  • plugins/wayback/PARSER_HOWTO.md

 80

 82- `target_url`: the exact URL used to find this capture
 83- `target_type`: `page` or `file`

 97

134    def http_get(
135        self,
136        url: str,
137        *,
138        timeout: int | None = None,
139        headers: dict[str, str] | None = None,
140    ) -> bytes:
141        """Fetch one URL with retries and return raw response bytes."""
142
143        timeout = self.timeout if timeout is None else timeout
144        last_error: Exception | None = None
145        for attempt in range(1, self.retries + 1):
146            request = build_request(url, extra_headers=headers)
147            try:
148                self.log(f"[http] GET attempt {attempt}/{self.retries}: {url}")
149                with urllib.request.urlopen(request, timeout=timeout) as response:
150                    payload = response.read()
151                    self.log(f"[http] GET ok ({len(payload)} bytes): {url}")
152                    return payload
153            except (urllib.error.HTTPError, urllib.error.URLError, TimeoutError) as exc:
154                last_error = exc
155                self.log(f"[http] GET failed: {url} -> {exc}")
156                if attempt == self.retries:
157                    break
158                time.sleep(attempt)
159        assert last_error is not None
160        raise last_error

162    def http_stream(self, url: str, destination: Path, *, timeout: int | None = None) -> None:
163        """Stream one URL into a local file with retries."""
164
165        timeout = self.timeout if timeout is None else timeout
166        destination.parent.mkdir(parents=True, exist_ok=True)
167        temp_path = destination.with_suffix(destination.suffix + ".part")
168        last_error: Exception | None = None
169        for attempt in range(1, self.retries + 1):
170            request = build_request(url)
171            try:
172                self.log(f"[http] STREAM attempt {attempt}/{self.retries}: {url} -> {destination}")
173                bytes_written = 0
174                next_progress = 25 * 1024 * 1024
175                with urllib.request.urlopen(request, timeout=timeout) as response, temp_path.open("wb") as handle:
176                    while True:
177                        chunk = response.read(1024 * 1024)
178                        if not chunk:
179                            break
180                        handle.write(chunk)
181                        bytes_written += len(chunk)
182                        if self.verbose and bytes_written >= next_progress:
183                            print(f"[http] STREAM progress {bytes_written} bytes: {destination}", flush=True)
184                            next_progress += 25 * 1024 * 1024
185                temp_path.replace(destination)
186                self.log(f"[http] STREAM ok ({bytes_written} bytes): {destination}")
187                return
188            except (urllib.error.HTTPError, urllib.error.URLError, TimeoutError) as exc:
189                last_error = exc
190                self.log(f"[http] STREAM failed: {url} -> {exc}")
191                if temp_path.exists():
192                    temp_path.unlink()
193                if attempt == self.retries:
194                    break
195                time.sleep(attempt * 2)
196        assert last_error is not None
197        raise last_error

205        self,

208        fl: tuple[str, ...],
209        match_type: str | None = None,
210        filters: tuple[str, ...] = (),
211        collapse: str | None = None,

214        """Execute one CDX query and return the raw JSON rows."""

216        params: list[tuple[str, str]] = [("url", url), ("output", "json"), ("fl", ",".join(fl))]

273    @staticmethod
274    def nearest_capture(captures: list[Capture], timestamp: str | None = None) -> Capture | None:
275        """Choose the nearest capture in time from an existing capture list."""
276
277        if not captures:
278            return None
279        if not timestamp:
280            return captures[-1]
281        return min(captures, key=lambda capture: (abs(int(capture.timestamp) - int(timestamp)), capture.timestamp))

283    def strict_file_capture(self, url: str, timestamp: str | None) -> Capture | None:
284        """Return the exact-timestamp capture for one file URL when present."""
285
286        if not timestamp:
287            return None
288        captures = self.exact_file_captures(url, collapse=None)
289        for capture in captures:
290            if capture.timestamp == timestamp:
291                return capture

294    def closest_page_capture(self, url: str, timestamp: str | None = None, *, policy: str = "nearest") -> Capture | None:
295        """Resolve the archived capture FAT should use for one binary or page URL."""
296
297        if policy == "strict":
298            return self.strict_file_capture(url, timestamp)
299        params = {"url": url}
300        if timestamp:
301            params["timestamp"] = timestamp
302        query = f"{self.availability_api}?{urllib.parse.urlencode(params)}"
303        payload = self.fetch_json(query)
304        archived = payload.get("archived_snapshots", {})  # type: ignore[assignment]
305        closest = archived.get("closest") if isinstance(archived, dict) else None
306        if not isinstance(closest, dict):
307            return self.nearest_capture(self.exact_file_captures(url), timestamp)
308        if str(closest.get("available")).lower() != "true":
309            return self.nearest_capture(self.exact_file_captures(url), timestamp)
310        return Capture(timestamp=str(closest["timestamp"]), original=url, statuscode=str(closest["status"]))

322    def fetch_snapshot_html(self, capture: Capture) -> str:

331    def fetch_snapshot_html_rendered(
332        self,
333        capture: Capture,
334        *,
335        mode: str = "required",
336        wait_until: str = "domcontentloaded",
337        wait_for: str | None = None,
338        timeout: int | None = None,
339    ) -> str:
340        """Fetch one archived HTML page through an isolated browser session.
341
342        The browser always navigates to the archived replay URL. JSON and
343        other non-page captures retain the direct HTTP behavior because they
344        are data resources rather than browser documents.
345        """
346
347        mimetype = str(capture.mimetype or "").lower()
348        if "json" in mimetype or mode == "static":
349            return self.fetch_snapshot_html(capture)
350        options = BrowserOptions(
351            mode=mode,
352            wait_until=wait_until,
353            wait_for=wait_for,
354            timeout=int(timeout or self.timeout),
355        )
356        replay_url = self.replay_page_url(capture)
357        try:
358            with BrowserSession(options) as browser:
359                return browser.render(replay_url, timeout=options.timeout)
360        except BrowserError:
361            if mode != "preferred":
362                raise
363            self.log(f"[browser] rendering failed; falling back to static HTML: {replay_url}")
364            return self.fetch_snapshot_html(capture)

377    def fetch_snapshot_html_near(
378        self,
379        url: str,
380        timestamp: str,
381        *,
382        render_mode: str | None = None,
383        wait_until: str | None = None,
384        wait_for: str | None = None,
385        timeout: int | None = None,
386    ) -> tuple[str | None, Capture | None]:
387        """Fetch archived page HTML at or near one timestamp.
388
389        This tries the exact timestamp first, then falls back to the nearest
390        available capture. Parsers commonly use this when a page links to a
391        second-level archived download page.
392        """
393
394        active = self._snapshot_render_options
395        mode = active.mode if render_mode is None else render_mode
396        wait_strategy = active.wait_until if wait_until is None else wait_until
397        selector = active.wait_for if wait_for is None else wait_for
398        render_timeout = active.timeout if timeout is None else timeout
399
400        def fetch(capture: Capture) -> str:
401            if mode == "static":
402                return self.fetch_snapshot_html(capture)
403            return self.fetch_snapshot_html_rendered(
404                capture,
405                mode=mode,
406                wait_until=wait_strategy,
407                wait_for=selector,
408                timeout=render_timeout,
409            )
410
411        try:
412            capture = Capture(timestamp=timestamp, original=url, statuscode="200")
413            return fetch(capture), capture
414        except (urllib.error.HTTPError, urllib.error.URLError, TimeoutError):
415            pass
416
417        capture = self.closest_page_capture(url, timestamp)
418        if capture is None:
419            return (None, None)
420        try:
421            return fetch(capture), capture
422        except (urllib.error.HTTPError, urllib.error.URLError, TimeoutError):
423            return (None, None)

425    def download_capture(self, capture: Capture, destination: Path) -> Path:

def http_stream( self, url: str, destination: pathlib.Path, *, timeout: int | None = None) -> None: 
162    def http_stream(self, url: str, destination: Path, *, timeout: int | None = None) -> None:
163        """Stream one URL into a local file with retries."""
164
165        timeout = self.timeout if timeout is None else timeout
166        destination.parent.mkdir(parents=True, exist_ok=True)
167        temp_path = destination.with_suffix(destination.suffix + ".part")
168        last_error: Exception | None = None
169        for attempt in range(1, self.retries + 1):
170            request = build_request(url)
171            try:
172                self.log(f"[http] STREAM attempt {attempt}/{self.retries}: {url} -> {destination}")
173                bytes_written = 0
174                next_progress = 25 * 1024 * 1024
175                with urllib.request.urlopen(request, timeout=timeout) as response, temp_path.open("wb") as handle:
176                    while True:
177                        chunk = response.read(1024 * 1024)
178                        if not chunk:
179                            break
180                        handle.write(chunk)
181                        bytes_written += len(chunk)
182                        if self.verbose and bytes_written >= next_progress:
183                            print(f"[http] STREAM progress {bytes_written} bytes: {destination}", flush=True)
184                            next_progress += 25 * 1024 * 1024
185                temp_path.replace(destination)
186                self.log(f"[http] STREAM ok ({bytes_written} bytes): {destination}")
187                return
188            except (urllib.error.HTTPError, urllib.error.URLError, TimeoutError) as exc:
189                last_error = exc
190                self.log(f"[http] STREAM failed: {url} -> {exc}")
191                if temp_path.exists():
192                    temp_path.unlink()
193                if attempt == self.retries:
194                    break
195                time.sleep(attempt * 2)
196        assert last_error is not None
197        raise last_error
